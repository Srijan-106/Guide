# Guide - Python dependencies
#
# Easiest install:
#
#   python setup.py
#
# It checks your system, then installs everything below. To install by
# hand instead:
#
#   pip install -r requirements.txt
#
# Versions are pinned (==) to the exact versions the project was built
# and tested with, so a new install behaves the same way.
#
# ---------------------------------------------------------------------
# PYTORCH - install separately first
# ---------------------------------------------------------------------
# torch and torchvision are not listed here, because the GPU build comes
# from PyTorch's own index rather than PyPI:
#
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu128
#
# Install it BEFORE this file: openai-whisper and transformers depend on
# torch, and if it is missing pip would fetch the CPU-only build instead.
#
# This stable build supports current NVIDIA GPUs including the RTX
# 50-series.
#
# Setup guide:
#   https://github.com/Srijan-106/Guide/blob/main/docs/setup.md
#
# Troubleshooting (RTX 50-series, Windows Smart App Control):
#   https://github.com/Srijan-106/Guide/blob/main/docs/gpu-setup.md
#
# Official PyTorch install selector:
#   https://pytorch.org/get-started/locally/
# ---------------------------------------------------------------------


# ===== Core: seeing and reading =======================================

# Object detection (YOLO)
ultralytics==8.4.155

# Camera capture, image handling, display windows
opencv-python==5.0.0.93

# Optical Character Recognition - reading text from images
easyocr==1.7.2

# Scientific computing, required by EasyOCR.
# Pinned below 1.18 because 1.18.1 was blocked by Windows Smart App
# Control on the development machine. See docs/gpu-setup.md.
scipy>=1.17,<1.18

# Spelling correction for words EasyOCR read with low confidence.
# Optional: without it, text is spoken exactly as EasyOCR read it.
pyspellchecker==0.9.0

# Numerical arrays and image processing
numpy==2.5.2
pillow==12.3.0


# ===== Core: speaking =================================================

# Speech: edge-tts generates the audio, pygame plays it.
# Requires an internet connection.
# 7.3.1, not 7.2.8: Microsoft changed the security token its voice
# service requires, and on 19 October 2026 7.2.8 was refused ("403")
# on every sentence. See docs/devlog.md, Day 16.
edge-tts==7.3.1
pygame==2.6.1

# Offline speech fallback, used only when edge-tts cannot reach the
# network. It is a fallback and not the primary engine because it failed
# five separate times when used as the main engine - see docs/devlog.md
# (Days 5, 5A and 8) for what changed to make it safe here.
pyttsx3==2.99


# ===== On-screen display (Hindi text) =================================

# Pillow on Windows usually cannot join Hindi letters correctly (its
# Raqm text-layout support is missing). These two shape the text
# instead: HarfBuzz decides how letters combine, FreeType draws them.
uharfbuzz==0.56.2
freetype-py==2.5.1


# ===== Optional: voice commands (V key) ===============================
# The app still runs without these; voice commands are switched off.

# Speech recognition (Whisper "small", runs locally on the GPU).
# Downloads about 460 MB of model weights on first use.
# Note: it installs llvmlite, whose DLL was blocked by Windows Smart App
# Control on the development machine. See docs/gpu-setup.md.
openai-whisper==20250625

# Microphone input
sounddevice==0.5.6


# ===== Optional: visual questions =====================================
# The app still runs without these; questions are switched off.

# Python client for Ollama. Ollama itself is a separate program,
# installed from https://ollama.com - then download the model once:
#
#   ollama pull qwen3-vl:4b-instruct
#
ollama==0.6.3


# ===== Optional: Hindi translation ====================================
# The app still runs without these: Hindi mode keeps its fixed Hindi
# phrases, but answers to visual questions are spoken in English.

# Runs the NLLB-200 translation model (English to Hindi) on the CPU.
# Downloads about 2.4 GB on first use of Hindi mode.
# NLLB-200 is licensed CC-BY-NC 4.0: free for non-commercial use only.
transformers==5.18.0

# Needed by the NLLB tokenizer, which splits text into model tokens
sentencepiece==0.2.2
